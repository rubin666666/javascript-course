💡 Приклади для тестування:

1. Змінні та типи даних
let name = "Анна"; console.log(typeof name);
2. Арифметичні операції
let a = 10, b = 3; console.log(a + b, a * b);
3. Умовні конструкції
let age = 20; if(age >= 18) console.log("Дорослий");
4. Цикли
for(let i = 1; i <= 5; i++) console.log(i);
5. Функції
function greet(name) { return "Привіт, " + name; }
6. Масиви
let arr = [1,2,3]; console.log(arr.map(x => x*2));
7. Об'єкти
let user = {name: "Іван", age: 25}; console.log(user.name);
8. DOM Basics
console.log(document.querySelectorAll('div').length);
9. Events
document.addEventListener('click', () => console.log('click'));
10. Fetch API
fetch('https://jsonplaceholder.typicode.com/todos/1').then(r=>r.json()).then(console.log);
